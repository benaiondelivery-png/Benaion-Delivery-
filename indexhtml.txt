<script>
    function irPara(telaId) {
        // Esconde todas as telas
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        // Mostra a tela desejada
        document.getElementById(telaId).classList.add('active');
    }

    // Auto-avançar da Splash para Perfil após 3 segundos
    setTimeout(() => {
        if(document.getElementById('splash').classList.contains('active')) {
            irPara('escolha-perfil');
        }
    }, 3000);

<script>
    function enviarWhatsApp() {
        const telefone = "5592991234567"; // SUBSTITUA PELO SEU NÚMERO (Ex: 55 + DDD + Numero)
        
        // Captura os valores dos campos
        const cliente = document.getElementById('cliente').value;
        const retirada = document.getElementById('retirada').value;
        const produto = document.getElementById('produto').value;
        const entrega = document.getElementById('entrega').value;
        const pagamento = document.getElementById('pagamento').value;
        const troco = document.getElementById('troco').value || "Não necessário";
        const agendar = document.getElementById('agendar').value || "Imediato";
        const obs = document.getElementById('obs').value || "Nenhuma";

        // Monta a mensagem com a sua formatação exata
        const mensagem = 
`*✅️COMANDA DE PEDIDO🗒*

> _*CLIENTE🙋‍♂️*_: ${cliente}

> *ENTREGADOR🛵 :* Gabriel (Benaion)

> _*ENDEREÇO DE RETIRADA🛒*_: ${retirada}

> _*QUAL PRODUTO* _: ${produto}

> _*ENDEREÇO DE ENTREGA (bairro, rua, numero)🏡*_: ${entrega}

> _*FORMA DE PAGAMENTO💰*_: ${pagamento}

> _*TROCO💱*_: ${troco}

> _*Agendar para às*_: ${agendar}
    
> *OBSERVAÇÕES:* ${obs}

*<<<<FIM DE COMANDA>>>>*`;

        // Codifica a mensagem para URL
        const url = "https://wa.me/" + telefone + "?text=" + encodeURIComponent(mensagem);
        
        // Abre o WhatsApp
        window.open(url, '_blank');
    }
